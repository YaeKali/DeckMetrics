<script>
(function(){
  "use strict";

  const STORAGE_KEY = "vaultArcana.v1";
  const API = "https://api.scryfall.com";

  let state = loadState();
  let activeDeckId = state.activeDeckId || null;
  let searchAbort = null;
  let searchDebounce = null;

  function loadState(){
    try{
      const raw = localStorage.getItem(STORAGE_KEY);
      if(raw) return JSON.parse(raw);
    }catch(e){ console.warn("localStorage unavailable", e); }
    return { decks:{}, deckOrder:[], activeDeckId:null };
  }

  function saveState(){
    state.activeDeckId = activeDeckId;
    try{
      localStorage.setItem(STORAGE_KEY, JSON.stringify(state));
    }catch(e){
      toast("Couldn't save locally — export your deck as backup.");
    }
  }

  function uid(){ return Date.now().toString(36) + Math.random().toString(36).slice(2,8); }

  function toast(msg){
    const t = document.getElementById("toast");
    t.textContent = msg;
    t.classList.add("show");
    clearTimeout(t._timer);
    t._timer = setTimeout(()=> t.classList.remove("show"), 2400);
  }

  function createDeck(name){
    const id = uid();
    state.decks[id] = { id, name: name || "Untitled deck", cards:{}, lastUpdated:null, cover:null };
    state.deckOrder.push(id);
    saveState();
    return id;
  }

  function cardImageUrl(card, size){
    size = size || "normal";
    if(card.image_uris) return card.image_uris[size] || card.image_uris.normal || "";
    if(card.card_faces && card.card_faces[0] && card.card_faces[0].image_uris){
      return card.card_faces[0].image_uris[size] || card.card_faces[0].image_uris.normal || "";
    }
    return "";
  }

  function colorDotsHtml(card){
    const colors = card.colors || (card.card_faces && card.card_faces[0] && card.card_faces[0].colors) || [];
    if(!colors.length) return '<span class="color-dots"><span class="color-dot dot-c"></span></span>';
    const map = { W:"dot-w", U:"dot-u", B:"dot-b", R:"dot-r", G:"dot-g" };
    return '<span class="color-dots">' + colors.map(c=>`<span class="color-dot ${map[c]||"dot-c"}"></span>`).join("") + '</span>';
  }

  function showRailPreview(card){
    const img = document.getElementById("previewRailImg");
    const ph = document.getElementById("previewRailPlaceholder");
    const price = cardPrice(card);
    img.src = cardImageUrl(card, "normal");
    img.classList.add("show");
    ph.classList.add("hide");
    document.getElementById("previewRailPrice").textContent = price !== null ? "$"+price.toFixed(2) : "";
    document.getElementById("previewRailName").textContent = card.name;
  }

  function setDeckCover(deckId, card){
    const deck = state.decks[deckId];
    if(!deck) return;
    const img = cardImageUrl(card, "art_crop");
    if(!img){ toast("No art available for that card."); return; }
    deck.cover = { image: img, name: card.name };
    saveState();
    renderAll();
    toast(`Cover set to ${card.name}`);
  }

  function clearDeckCover(deckId){
    const deck = state.decks[deckId];
    if(!deck) return;
    deck.cover = null;
    saveState();
    renderAll();
  }

  function getActiveDeck(){
    return activeDeckId ? state.decks[activeDeckId] : null;
  }

  function typeGroup(typeLine){
    if(!typeLine) return "Other";
    const t = typeLine.toLowerCase();
    if(t.includes("creature")) return "Creatures";
    if(t.includes("planeswalker")) return "Planeswalkers";
    if(t.includes("battle")) return "Battles";
    if(t.includes("instant")) return "Instants";
    if(t.includes("sorcery")) return "Sorceries";
    if(t.includes("artifact")) return "Artifacts";
    if(t.includes("enchantment")) return "Enchantments";
    if(t.includes("land")) return "Lands";
    return "Other";
  }
  const GROUP_ORDER = ["Creatures","Planeswalkers","Instants","Sorceries","Artifacts","Enchantments","Battles","Lands","Other"];

  function cardPrice(cardData){
    if(!cardData || !cardData.prices) return null;
    const usd = cardData.prices.usd;
    return usd ? parseFloat(usd) : null;
  }

  function deckTotal(deck){
    let total = 0, unpriced = 0;
    Object.values(deck.cards).forEach(entry=>{
      const p = cardPrice(entry.cardData);
      if(p !== null){ total += p * entry.qty; }
      else unpriced += entry.qty;
    });
    return { total, unpriced };
  }

  function renderDeckList(){
    const el = document.getElementById("deckList");
    el.innerHTML = "";
    if(state.deckOrder.length === 0){
      el.innerHTML = '<div class="empty-hint">No decks yet. Create one to start adding cards.</div>';
      return;
    }
    state.deckOrder.forEach(id=>{
      const deck = state.decks[id];
      if(!deck) return;
      const count = Object.values(deck.cards).reduce((s,c)=>s+c.qty,0);
      const div = document.createElement("div");
      div.className = "deck-item" + (id === activeDeckId ? " active" : "");
      const hasCover = deck.cover && deck.cover.image;
      div.innerHTML = `
        <div class="deck-cover${hasCover ? " has-image" : ""}" style="${hasCover ? `background-image:url('${deck.cover.image}')` : ""}">
          <span class="glyph">✦</span>
          ${hasCover ? `<button class="deck-cover-clear" data-clear-cover="${id}" title="Remove cover">✕</button>` : ""}
        </div>
        <div class="text">
          <div class="name">${escapeHtml(deck.name)}</div>
          <div class="meta">${count} card${count===1?"":"s"}</div>
        </div>`;
      div.addEventListener("click", ()=>{ activeDeckId = id; saveState(); renderAll(); });
      const clearBtn = div.querySelector("[data-clear-cover]");
      if(clearBtn){
        clearBtn.addEventListener("click", ev=>{ ev.stopPropagation(); clearDeckCover(id); });
      }
      el.appendChild(div);
    });
  }

  function renderDeckEditor(){
    const empty = document.getElementById("deckEditorEmpty");
    const editor = document.getElementById("deckEditor");
    const deck = getActiveDeck();
    if(!deck){
      empty.classList.remove("hidden");
      editor.classList.add("hidden");
      return;
    }
    empty.classList.add("hidden");
    editor.classList.remove("hidden");
    editor.style.display = "flex";

    document.getElementById("deckTitle").textContent = deck.name;
    const { total, unpriced } = deckTotal(deck);
    document.getElementById("deckTotal").textContent = "$" + total.toFixed(2);
    const count = Object.values(deck.cards).reduce((s,c)=>s+c.qty,0);
    document.getElementById("deckCount").textContent =
      count + " card" + (count===1?"":"s") + (unpriced ? ` · ${unpriced} unpriced` : "");
    document.getElementById("lastUpdated").textContent = deck.lastUpdated
      ? "priced " + timeAgo(deck.lastUpdated)
      : "never priced";

    const body = document.getElementById("deckBody");
    body.innerHTML = "";
    const statsBar = document.getElementById("statsBar");
    const entries = Object.values(deck.cards);
    updateAnalytics(deck);
    if(entries.length === 0){
      body.innerHTML = '<div class="empty-hint">This deck is empty. Search for cards on the left and add them here.</div>';
      statsBar.innerHTML = "";
      return;
    }

    const groups = {};
    entries.forEach(e=>{
      const g = typeGroup(e.cardData.type_line);
      (groups[g] = groups[g] || []).push(e);
    });

    GROUP_ORDER.forEach(g=>{
      if(!groups[g]) return;
      const list = groups[g].sort((a,b)=> a.cardData.name.localeCompare(b.cardData.name));
      const gCount = list.reduce((s,c)=>s+c.qty,0);
      const wrap = document.createElement("div");
      wrap.className = "group";
      wrap.innerHTML = `<div class="group-title"><span>${g}</span><span>${gCount}</span></div>`;
      list.forEach(entry=>{
        const price = cardPrice(entry.cardData);
        const thumb = cardImageUrl(entry.cardData, "small");
        const row = document.createElement("div");
        row.className = "deck-card-row";
        row.innerHTML = `
          <div class="qty-controls">
            <button data-act="dec" data-id="${entry.cardData.id}">−</button>
            <span class="qty-num">${entry.qty}</span>
            <button data-act="inc" data-id="${entry.cardData.id}">+</button>
          </div>
          ${thumb ? `<img class="deck-card-thumb" data-act="preview" data-id="${entry.cardData.id}" src="${thumb}" alt="">` : `<div class="deck-card-thumb" data-act="preview" data-id="${entry.cardData.id}"></div>`}
          <div class="deck-card-name" data-act="preview" data-id="${entry.cardData.id}">${escapeHtml(entry.cardData.name)}</div>
          <div class="deck-card-price">${colorDotsHtml(entry.cardData)}${price !== null ? "$"+price.toFixed(2) : "—"}</div>
          <button class="remove-x" data-act="remove" data-id="${entry.cardData.id}" title="Remove">✕</button>
        `;
        row.addEventListener("mouseenter", ()=> showRailPreview(entry.cardData));
        wrap.appendChild(row);
      });
      body.appendChild(wrap);
    });

    statsBar.innerHTML = GROUP_ORDER
      .filter(g=> groups[g])
      .map(g=> `<span class="stat"><b>${groups[g].reduce((s,c)=>s+c.qty,0)}</b> ${g}</span>`)
      .join("");

    body.querySelectorAll("[data-act]").forEach(el2=>{
      el2.addEventListener("click", ()=>{
        const id = el2.dataset.id, act = el2.dataset.act;
        const entry = deck.cards[id];
        if(!entry) return;
        if(act === "inc") { entry.qty += 1; saveState(); renderAll(); }
        if(act === "dec"){ entry.qty -= 1; if(entry.qty <= 0) delete deck.cards[id]; saveState(); renderAll(); }
        if(act === "remove"){ delete deck.cards[id]; saveState(); renderAll(); }
        if(act === "preview") openCardPreview(entry.cardData);
      });
    });
  }
function updateAnalytics(deck) {
    const panel = document.getElementById("analyticsPanel");
    const entries = Object.values(deck.cards);
    
    if (entries.length === 0) {
      panel.style.display = "none";
      return;
    }
    panel.style.display = "flex";
  
    // Variables for math
    const curve = [0, 0, 0, 0, 0, 0, 0];
    let maxCurve = 0;
    const pips = { W: 0, U: 0, B: 0, R: 0, G: 0 };
    let totalPips = 0;
  
    entries.forEach(entry => {
      const card = entry.cardData;
      const qty = entry.qty;
      const isLand = card.type_line && card.type_line.toLowerCase().includes("land");
  
      if (!isLand) {
        // Curve Math (Cap at 6+)
        const cmc = Math.min(Math.floor(card.cmc || 0), 6);
        curve[cmc] += qty;
        if (curve[cmc] > maxCurve) maxCurve = curve[cmc];
  
        // Color Pip Math
        const costStr = card.mana_cost || "";
        const matches = costStr.match(/{[WUBRG]}/g);
        if (matches) {
          matches.forEach(m => {
            const color = m[1]; // grabs the letter inside { }
            pips[color] += qty;
            totalPips += qty;
          });
        }
      }
    });
  
    document.getElementById("manaCurveChart").innerHTML = curve.map(val => {
      const height = maxCurve > 0 ? (val / maxCurve) * 100 : 0;
      return `
        <div style="flex:1; display:flex; flex-direction:column; justify-content:flex-end; align-items:center;">
          <span style="font-size:9px; color:var(--muted); margin-bottom:2px;">${val || ''}</span>
          <div style="width:100%; height:${height}%; background:var(--gold); border-radius:2px 2px 0 0; opacity:${val > 0 ? 1 : 0.1}; transition:height 0.3s;"></div>
        </div>
      `;
    }).join("");
  
    const colors = [
      { key: 'W', hex: '#f0e6c8' }, { key: 'U', hex: '#4f8fd9' },
      { key: 'B', hex: '#9a8fc0' }, { key: 'R', hex: '#d1543a' }, { key: 'G', hex: '#4f9d5e' }
    ];
    document.getElementById("colorBar").innerHTML = totalPips === 0 
      ? '<div style="flex:1; display:flex; align-items:center; justify-content:center; font-size:10px; color:var(--muted);">No colored mana</div>'
      : colors.map(c => {
          const count = pips[c.key];
          if (count === 0) return '';
          const pct = (count / totalPips) * 100;
          return `<div style="width:${pct}%; background:${c.hex}; display:flex; justify-content:center; align-items:center; font-size:10px; color:#000; font-weight:bold;" title="${count} ${c.key}">${count}</div>`;
        }).join("");
  
    const sortedByPrice = entries
      .map(e => ({ name: e.cardData.name, cost: (cardPrice(e.cardData) || 0) * e.qty }))
      .sort((a, b) => b.cost - a.cost)
      .filter(e => e.cost > 0)
      .slice(0, 3); // Top 3 limit
  
    document.getElementById("topValueList").innerHTML = sortedByPrice.length === 0
      ? '<span>No priced cards</span>'
      : sortedByPrice.map(item => `
          <div style="display:flex; justify-content:space-between;">
            <span style="white-space:nowrap; overflow:hidden; text-overflow:ellipsis;">${escapeHtml(item.name)}</span>
            <span style="color:var(--gold-bright); font-variant-numeric:tabular-nums;">$${item.cost.toFixed(2)}</span>
          </div>
        `).join("");
  }

  function timeAgo(iso){
    const diff = Date.now() - new Date(iso).getTime();
    const mins = Math.floor(diff/60000);
    if(mins < 1) return "just now";
    if(mins < 60) return mins + "m ago";
    const hrs = Math.floor(mins/60);
    if(hrs < 24) return hrs + "h ago";
    return Math.floor(hrs/24) + "d ago";
  }

  function escapeHtml(s){
    return String(s).replace(/[&<>"']/g, c=>({ "&":"&amp;","<":"&lt;",">":"&gt;",'"':"&quot;","'":"&#39;" }[c]));
  }

  function renderAll(){
    renderDeckList();
    renderDeckEditor();
  }

  function runSearch(q){
    const resultsEl = document.getElementById("searchResults");
    if(!q || q.trim().length < 2){
      resultsEl.innerHTML = '<div class="empty-hint">Type a card name — e.g. <b>Sol Ring</b> or <b>Lightning Bolt</b> — and click a result to add it to the open deck.</div>';
      return;
    }
    resultsEl.innerHTML = '<div class="empty-hint"><span class="spinner"></span>&nbsp; searching…</div>';
    if(searchAbort) searchAbort.abort();
    searchAbort = new AbortController();
    let query = q;
    if (activeSearchType) query += ` t:${activeSearchType}`;
    fetch(`${API}/cards/search?q=${encodeURIComponent(query)}&order=name`, { signal: searchAbort.signal })
      .then(r=>{
        if(r.status === 404) return { data: [] };
        if(!r.ok) throw new Error("search failed");
        return r.json();
      })
      .then(json=>{
        setNet(true);
        const cards = (json.data || []).slice(0, 60);
        resultsEl.innerHTML = "";
        if(cards.length === 0){
          resultsEl.innerHTML = '<div class="empty-hint">No cards found for that name.</div>';
          return;
        }

        function buildRow(card){
          const img = cardImageUrl(card, "small");
          const price = cardPrice(card);
          const row = document.createElement("div");
          row.className = "card-row";
          row.innerHTML = `
            ${img ? `<img class="card-thumb" src="${img}" alt="">` : `<div class="card-thumb"></div>`}
            <div class="card-info">
              <div class="card-name">${escapeHtml(card.name)}</div>
              <div class="card-sub">${escapeHtml(card.set_name || "")} · ${escapeHtml((card.type_line||"").split(" — ")[0])}</div>
            </div>
            <div class="card-price">${colorDotsHtml(card)}${price !== null ? "$"+price.toFixed(2) : "—"}</div>
          `;
          const thumbEl = row.querySelector(".card-thumb");
          thumbEl.addEventListener("click", ev=>{ ev.stopPropagation(); openCardPreview(card); });
          row.addEventListener("mouseenter", ()=> showRailPreview(card));
          row.addEventListener("click", ()=> addCardToActiveDeck(card));
          return row;
        }

        const groups = {};
        cards.forEach(card=>{
          const tl = card.type_line || (card.card_faces && card.card_faces[0] && card.card_faces[0].type_line) || "";
          const g = typeGroup(tl);
          (groups[g] = groups[g] || []).push(card);
        });

        GROUP_ORDER.forEach(g=>{
          if(!groups[g]) return;
          const wrap = document.createElement("div");
          wrap.className = "group";
          wrap.innerHTML = `<div class="group-title"><span>${g}</span><span>${groups[g].length}</span></div>`;
          groups[g].forEach(card=> wrap.appendChild(buildRow(card)));
          resultsEl.appendChild(wrap);
        });
      })
      .catch(err=>{
        if(err.name === "AbortError") return;
        setNet(false);
        resultsEl.innerHTML = '<div class="empty-hint">Couldn\'t reach Scryfall. Check your connection and try again.</div>';
      });
  }

  function addCardToActiveDeck(card){
    const deck = getActiveDeck();
    if(!deck){ toast("Create or select a deck first."); return; }
    const existing = deck.cards[card.id];
    if(existing) existing.qty += 1;
    else deck.cards[card.id] = { qty:1, cardData: card };
    saveState();
    renderAll();
    toast(`Added ${card.name}`);
  }

  function refreshDeckPrices(){
    const deck = getActiveDeck();
    if(!deck) return;
    const ids = Object.keys(deck.cards);
    if(ids.length === 0){ toast("Deck is empty."); return; }

    const btn = document.getElementById("refreshBtn");
    btn.disabled = true;
    btn.innerHTML = '<span class="spinner"></span> refreshing…';

    const batches = [];
    for(let i=0;i<ids.length;i+=75) batches.push(ids.slice(i,i+75));

    Promise.all(batches.map(batch=>
      fetch(`${API}/cards/collection`, {
        method:"POST",
        headers:{ "Content-Type":"application/json" },
        body: JSON.stringify({ identifiers: batch.map(id=>({ id })) })
      }).then(r=>{
        if(!r.ok) throw new Error("collection fetch failed");
        return r.json();
      })
    )).then(results=>{
      setNet(true);
      results.forEach(json=>{
        (json.data || []).forEach(card=>{
          if(deck.cards[card.id]) deck.cards[card.id].cardData = card;
        });
      });
      deck.lastUpdated = new Date().toISOString();
      saveState();
      renderAll();
      toast("Prices updated.");
    }).catch(()=>{
      setNet(false);
      toast("Couldn't refresh prices — check your connection.");
    }).finally(()=>{
      btn.disabled = false;
      btn.textContent = "Refresh prices";
    });
  }

  let currentPreviewCard = null;

  function openCardPreview(card){
    currentPreviewCard = card;
    const img = cardImageUrl(card, "normal");
    document.getElementById("previewImg").src = img;
    document.getElementById("previewImg").alt = card.name;
    document.getElementById("previewName").textContent = card.name;
    document.getElementById("previewCost").textContent = card.mana_cost || "";
    document.getElementById("previewType").textContent = card.type_line || "";
    document.getElementById("previewText").textContent =
      card.oracle_text || (card.card_faces ? card.card_faces.map(f=>f.oracle_text||"").filter(Boolean).join("\n\n") : "") || "";
    const pt = card.power !== undefined ? `${card.power}/${card.toughness}`
      : (card.loyalty !== undefined ? `Loyalty: ${card.loyalty}` : "");
    document.getElementById("previewPT").textContent = pt;
    document.getElementById("previewPT").style.display = pt ? "" : "none";
    const price = cardPrice(card);
    document.getElementById("previewPrice").textContent = price !== null ? "$"+price.toFixed(2) : "Price unavailable";

    const deck = getActiveDeck();
    const addBtn = document.getElementById("previewAddBtn");
    const coverBtn = document.getElementById("previewCoverBtn");
    addBtn.style.display = deck ? "" : "none";
    coverBtn.style.display = deck ? "" : "none";
    addBtn.textContent = "Add to " + (deck ? deck.name : "deck");

    document.getElementById("cardPreviewDialog").showModal();
  }

  document.getElementById("previewClose").addEventListener("click", ()=> document.getElementById("cardPreviewDialog").close());
  document.getElementById("previewAddBtn").addEventListener("click", ()=>{
    if(currentPreviewCard) addCardToActiveDeck(currentPreviewCard);
  });
  document.getElementById("previewCoverBtn").addEventListener("click", ()=>{
    if(currentPreviewCard && activeDeckId) setDeckCover(activeDeckId, currentPreviewCard);
  });

  function setNet(ok){
    const el = document.getElementById("netStatus");
    el.innerHTML = ok
      ? '<span class="dot"></span> connected'
      : '<span class="dot off"></span> offline';
  }

  function promptText(title, initial){
    return new Promise(resolve=>{
      const dlg = document.getElementById("promptDialog");
      document.getElementById("promptTitle").textContent = title;
      const input = document.getElementById("promptInput");
      input.value = initial || "";
      dlg.showModal();
      setTimeout(()=> input.focus(), 30);
      function cleanup(val){
        dlg.close();
        okBtn.removeEventListener("click", onOk);
        cancelBtn.removeEventListener("click", onCancel);
        input.removeEventListener("keydown", onKey);
        resolve(val);
      }
      const okBtn = document.getElementById("promptOk");
      const cancelBtn = document.getElementById("promptCancel");
      function onOk(){ cleanup(input.value.trim() || null); }
      function onCancel(){ cleanup(null); }
      function onKey(e){ if(e.key === "Enter"){ e.preventDefault(); onOk(); } if(e.key === "Escape") onCancel(); }
      okBtn.addEventListener("click", onOk);
      cancelBtn.addEventListener("click", onCancel);
      input.addEventListener("keydown", onKey);
    });
  }

  function groupDeckEntries(deck){
    const groups = {};
    Object.values(deck.cards).forEach(e=>{
      const g = typeGroup(e.cardData.type_line);
      (groups[g] = groups[g] || []).push(e);
    });
    Object.keys(groups).forEach(g=> groups[g].sort((a,b)=> a.cardData.name.localeCompare(b.cardData.name)));
    return groups;
  }

  function buildDeckText(deck){
    const { total } = deckTotal(deck);
    const count = Object.values(deck.cards).reduce((s,c)=>s+c.qty,0);
    const groups = groupDeckEntries(deck);
    let lines = [deck.name, `${count} cards · $${total.toFixed(2)} total`, ""];
    GROUP_ORDER.forEach(g=>{
      if(!groups[g]) return;
      lines.push(g + ":");
      groups[g].forEach(e=> lines.push(`${e.qty} ${e.cardData.name}`));
      lines.push("");
    });
    return lines.join("\n").trim() + "\n";
  }

  function exportDeckText(){
    const deck = getActiveDeck();
    if(!deck) return;
    const blob = new Blob([buildDeckText(deck)], { type:"text/plain" });
    const a = document.createElement("a");
    a.href = URL.createObjectURL(blob);
    a.download = (deck.name || "deck").replace(/[^a-z0-9\-_ ]/gi,"").trim().replace(/\s+/g,"-") + ".txt";
    document.body.appendChild(a);
    a.click();
    a.remove();
  }

  function copyDeckText(){
    const deck = getActiveDeck();
    if(!deck){ toast("Create or select a deck first."); return; }
    const text = buildDeckText(deck);
    if(navigator.clipboard && navigator.clipboard.writeText){
      navigator.clipboard.writeText(text).then(
        ()=> toast("Deck list copied."),
        ()=> fallbackCopy(text)
      );
    } else {
      fallbackCopy(text);
    }
  }

  function fallbackCopy(text){
    try{
      const ta = document.createElement("textarea");
      ta.value = text;
      ta.style.position = "fixed";
      ta.style.opacity = "0";
      document.body.appendChild(ta);
      ta.focus(); ta.select();
      document.execCommand("copy");
      ta.remove();
      toast("Deck list copied.");
    }catch(e){
      toast("Couldn't copy — try Download .txt instead.");
    }
  }

  function printDeck(){
    const deck = getActiveDeck();
    if(!deck){ toast("Create or select a deck first."); return; }
    const { total } = deckTotal(deck);
    const count = Object.values(deck.cards).reduce((s,c)=>s+c.qty,0);
    const groups = groupDeckEntries(deck);
    let html = `<h1>${escapeHtml(deck.name)}</h1><div class="print-meta">${count} cards · $${total.toFixed(2)} total · ${new Date().toLocaleDateString()}</div>`;
    GROUP_ORDER.forEach(g=>{
      if(!groups[g]) return;
      const gCount = groups[g].reduce((s,c)=>s+c.qty,0);
      html += `<h2>${g} (${gCount})</h2><table>`;
      groups[g].forEach(e=>{
        const p = cardPrice(e.cardData);
        html += `<tr><td>${e.qty}</td><td>${escapeHtml(e.cardData.name)}</td><td>${p!==null?"$"+p.toFixed(2):""}</td></tr>`;
      });
      html += "</table>";
    });
    document.getElementById("printArea").innerHTML = html;
    window.print();
  }

  function exportDeck(){
    const deck = getActiveDeck();
    if(!deck) return;
    const payload = {
      name: deck.name,
      exportedAt: new Date().toISOString(),
      cards: Object.values(deck.cards).map(e=>({ qty:e.qty, name:e.cardData.name, scryfallId:e.cardData.id }))
    };
    const blob = new Blob([JSON.stringify(payload, null, 2)], { type:"application/json" });

    a.remove();
  }

  function importDeckFile(file){
    const reader = new FileReader();
    reader.onload = ()=>{
      let payload;
      try{ payload = JSON.parse(reader.result); }
      catch(e){ toast("That file isn't valid deck JSON."); return; }
      if(!payload || !Array.isArray(payload.cards)){ toast("That file isn't valid deck JSON."); return; }
      const id = createDeck(payload.name || "Imported deck");
      const deck = state.decks[id];
      toast("Importing " + payload.cards.length + " card(s)…");
      const lookups = payload.cards.map(c=>
        c.scryfallId ? { id: c.scryfallId } : { name: c.name }
      );
      const batches = [];
      for(let i=0;i<lookups.length;i+=75) batches.push(lookups.slice(i,i+75));
      Promise.all(batches.map(batch=>
        fetch(`${API}/cards/collection`, {
          method:"POST",
          headers:{ "Content-Type":"application/json" },
          body: JSON.stringify({ identifiers: batch })
        }).then(r=>r.json())
      )).then(results=>{
        const byKey = {};
        results.forEach(json=> (json.data||[]).forEach(card=>{ byKey[card.name.toLowerCase()] = card; byKey[card.id] = card; }));
        payload.cards.forEach(c=>{
          const card = (c.scryfallId && byKey[c.scryfallId]) || byKey[(c.name||"").toLowerCase()];
          if(card) deck.cards[card.id] = { qty: c.qty || 1, cardData: card };
        });
        deck.lastUpdated = new Date().toISOString();
        activeDeckId = id;
        saveState();
        renderAll();
        toast("Deck imported.");
      }).catch(()=>{
        activeDeckId = id;
        saveState();
        renderAll();
        toast("Deck created, but prices couldn't be fetched.");
      });
    };
    reader.readAsText(file);
  }

  document.getElementById("importTextOk").addEventListener("click", () => {
    const text = document.getElementById("importTextInput").value;
    if (!text.trim()) return;
    document.getElementById("importTextDialog").close();

    const lines = text.split('\n').map(l => l.trim()).filter(Boolean);
    const lookups = [];
    const parsedCards = [];

    lines.forEach(line => {
      // Matches standard formats like "4 Lightning Bolt" or "1x Sol Ring"
      const match = line.match(/^(\d+)x?\s+(.+)$/i);
      let qty = 1;
      let name = line;
      if (match) {
        qty = parseInt(match[1], 10);
        name = match[2].trim();
      }
      // Remove set codes often found in MTGO exports e.g. "(MH2) 123"
      name = name.replace(/\s*\(.*?\)\s*\d*$/, '');
      lookups.push({ name: name });
      parsedCards.push({ qty, name });
    });

    if (!lookups.length) return;

    toast(`Importing ${lookups.length} unique cards...`);
    const id = createDeck("Imported from Text");
    const deck = state.decks[id];

    const batches = [];
    for(let i=0; i<lookups.length; i+=75) batches.push(lookups.slice(i, i+75));

    Promise.all(batches.map(batch =>
      fetch(`${API}/cards/collection`, {
        method: "POST",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify({ identifiers: batch })
      }).then(r => r.json())
    )).then(results => {
      const byKey = {};
      results.forEach(json => (json.data || []).forEach(card => {
        byKey[card.name.toLowerCase()] = card;
      }));

      parsedCards.forEach(c => {
        const card = byKey[c.name.toLowerCase()];
        if (card) {
          deck.cards[card.id] = { qty: c.qty, cardData: card };
        }
      });

      deck.lastUpdated = new Date().toISOString();
      activeDeckId = id;
      saveState();
      renderAll();
      toast("Text deck imported.");
    }).catch(() => {
      activeDeckId = id;
      saveState();
      renderAll();
      toast("Deck created, but some prices/cards couldn't be fetched.");
    });
  });

  let ptLibrary = [];
  let ptHand = [];

  function shuffle(array) {
    for (let i = array.length - 1; i > 0; i--) {
      const j = Math.floor(Math.random() * (i + 1));
      [array[i], array[j]] = [array[j], array[i]];
    }
    return array;
  }

  function initPlaytest() {
    const deck = getActiveDeck();
    if (!deck) { toast("Create or select a deck first."); return; }
    ptLibrary = [];
    ptHand = [];
    // Flatten the deck array so quantities are expanded into individual cards
    Object.values(deck.cards).forEach(entry => {
      for (let i = 0; i < entry.qty; i++) {
        ptLibrary.push(entry.cardData);
      }
    });
    shuffle(ptLibrary);
    document.getElementById("ptDeckName").textContent = deck.name;
    updatePlaytestUI();
    document.getElementById("playtestDialog").showModal();
  }

  function drawCards(num) {
    for (let i = 0; i < num; i++) {
      if (ptLibrary.length > 0) {
        ptHand.push(ptLibrary.shift());
      }
    }
    updatePlaytestUI();
  }

  function updatePlaytestUI() {
    document.getElementById("ptDeckCount").textContent = `${ptLibrary.length} in library`;
    const handContainer = document.getElementById("ptHand");
    handContainer.innerHTML = "";
    ptHand.forEach(card => {
      const imgUrl = cardImageUrl(card, "normal");
      if (imgUrl) {
        const img = document.createElement("img");
        img.className = "pt-card-img";
        img.src = imgUrl;
        handContainer.appendChild(img);
      }
    });
  }
